package handlers

import (
	"fmt"
	"net/http"
	"os"
	"path/filepath"
	"strings"
	"time"
	"unicode/utf8"

	"github.com/gin-gonic/gin"
	"gopkg.in/yaml.v3"

	// "your_project/models" // 替换为你的 models 包路径
	// "your_project/global" // 替换为你的 global 包路径 (假设 global.DB 是 gorm 实例)
)

// FrontMatter 定义 Markdown 头部的 YAML 结构
type FrontMatter struct {
	Title string   `yaml:"title"`
	Tags  []string `yaml:"tags"`
	// 可根据需要扩展其他字段，例如 date, description 等
}

// ---------------------------------------------------------
// 内部工具函数：解析 Markdown 文件
// ---------------------------------------------------------

// parseMarkdown 解析文件的 YAML Front Matter 并统计总字数
func parseMarkdown(filePath string) (title string, tags string, wordCount int, err error) {
	contentBytes, err := os.ReadFile(filePath)
	if err != nil {
		return "", "", 0, err
	}
	contentStr := string(contentBytes)

	// 1. 粗略统计字数 (中英文混合环境下，使用 RuneCount 比 len() 更准)
	wordCount = utf8.RuneCountInString(contentStr)

	// 2. 尝试提取并解析 YAML Front Matter
	// 标准的 Front Matter 必须位于文件最开头，并被 --- 包裹
	if strings.HasPrefix(contentStr, "---\n") || strings.HasPrefix(contentStr, "---\r\n") {
		// 按 "---" 分割字符串，最多分 3 份：[空字符串, YAML内容, 正文内容]
		parts := strings.SplitN(contentStr, "---", 3)
		if len(parts) >= 3 {
			yamlContent := parts[1]
			var fm FrontMatter
			// 解析 YAML 内容
			if err := yaml.Unmarshal([]byte(yamlContent), &fm); err == nil {
				title = strings.TrimSpace(fm.Title)
				// 将字符串切片转换为逗号分隔的字符串，存入 MySQL
				tags = strings.Join(fm.Tags, ",")
			}
		}
	}

	return title, tags, wordCount, nil
}

// ---------------------------------------------------------
// 路由接口实现
// ---------------------------------------------------------

// UploadDocumentHandler 处理 Markdown 文件上传与解析
func UploadDocumentHandler(c *gin.Context) {
	file, err := c.FormFile("file")
	if err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"error": "获取上传文件失败: " + err.Error()})
		return
	}

	// 1. 确保本地存储目录存在
	uploadDir := "./uploads/markdowns"
	if err := os.MkdirAll(uploadDir, os.ModePerm); err != nil {
		c.JSON(http.StatusInternalServerError, gin.H{"error": "创建存储目录失败"})
		return
	}

	// 2. 构造唯一文件保存路径 (时间戳 + 原文件名)
	newFileName := fmt.Sprintf("%d_%s", time.Now().Unix(), file.Filename)
	savePath := filepath.Join(uploadDir, newFileName)

	// 3. 将文件保存到本地硬盘
	if err := c.SaveUploadedFile(file, savePath); err != nil {
		c.JSON(http.StatusInternalServerError, gin.H{"error": "保存文件失败: " + err.Error()})
		return
	}

	// 4. 解析 Markdown 的元数据 (核心逻辑)
	title, tags, wordCount, parseErr := parseMarkdown(savePath)
	if parseErr != nil {
		// 即使解析失败，也不影响基本流程，记录日志即可
		fmt.Printf("解析文件 [%s] 元数据失败: %v\n", savePath, parseErr)
	}

	// 如果 Markdown 没有写 title，或者提取失败，则降级使用文件名（去除 .md 后缀）作为标题
	if title == "" {
		title = strings.TrimSuffix(file.Filename, filepath.Ext(file.Filename))
	}

	// 5. 构建数据库模型
	doc := models.Document{
		FileName:  file.Filename,
		Title:     title,
		FilePath:  savePath,
		Tags:      tags,
		WordCount: wordCount,
	}

	// 6. 开启 GORM 事务写入数据库
	// 使用事务是为了应对极端情况，虽然这里只有一条语句，但保持好习惯
	if err := global.DB.Create(&doc).Error; err != nil {
		// 数据库写入失败，执行数据补偿：删掉刚才保存的物理文件，防止产生脏文件
		_ = os.Remove(savePath)
		c.JSON(http.StatusInternalServerError, gin.H{"error": "数据库记录创建失败: " + err.Error()})
		return
	}

	// 7. 返回成功结果
	c.JSON(http.StatusOK, gin.H{
		"message": "文档上传并解析成功",
		"data":    doc,
	})
}

// DeleteDocumentHandler 处理文档的物理删除与逻辑删除
func DeleteDocumentHandler(c *gin.Context) {
	docID := c.Param("id")

	// 1. 去数据库查询记录，拿到 FilePath
	var doc models.Document
	if err := global.DB.First(&doc, docID).Error; err != nil {
		c.JSON(http.StatusNotFound, gin.H{"error": "未找到该文档记录"})
		return
	}

	// 2. 删除本地物理文件
	// os.IsNotExist 用于容错：如果物理文件已经被手动删了，不应该阻碍数据库记录的删除
	if err := os.Remove(doc.FilePath); err != nil && !os.IsNotExist(err) {
		c.JSON(http.StatusInternalServerError, gin.H{"error": "删除本地文件失败: " + err.Error()})
		return
	}

	// 3. 删除数据库记录 (GORM 软删除)
	if err := global.DB.Delete(&doc).Error; err != nil {
		c.JSON(http.StatusInternalServerError, gin.H{"error": "删除数据库记录失败: " + err.Error()})
		return
	}

	c.JSON(http.StatusOK, gin.H{
		"message":    "文档删除成功",
		"deleted_id": doc.ID,
	})
}
