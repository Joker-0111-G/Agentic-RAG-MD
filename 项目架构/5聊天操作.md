### 1. `listSessionsHandler` (获取左侧会话列表)

这个接口负责把历史对话按照**最后活跃时间（`updated_at`）倒序**排列并返回，让用户每次打开页面都能看到最近聊过的对话。

Go

```
package handlers

import (
	"net/http"

	"github.com/gin-gonic/gin"
	// "your_project/models" // 替换为你的 models 包路径
	// "your_project/global" // 替换为你的 global 包路径
)

func ListSessionsHandler(c *gin.Context) {
	var sessions []models.ChatSession

	// 按照 updated_at 倒序排列，把最近活跃的对话顶在最前面
	// 注意：如果未来扩展了多用户系统，这里需要加上 .Where("user_id = ?", currentUserID)
	if err := global.DB.Order("updated_at desc").Find(&sessions).Error; err != nil {
		c.JSON(http.StatusInternalServerError, gin.H{"error": "获取会话列表失败: " + err.Error()})
		return
	}

	c.JSON(http.StatusOK, gin.H{
		"message": "获取成功",
		"data":    sessions,
	})
}
```

### 2. `getSessionHistoryHandler` (获取某个会话的历史记录)

当用户在侧边栏点击某一个历史对话时，前端会调用这个接口，把当时的聊天上下文拉取出来渲染在主界面上。

Go

```
func GetSessionHistoryHandler(c *gin.Context) {
	sessionID := c.Param("session_id")

	if sessionID == "" {
		c.JSON(http.StatusBadRequest, gin.H{"error": "session_id 不能为空"})
		return
	}

	var history []models.ChatHistory

	// 按照创建时间升序排列，还原对话的真实先后顺序
	if err := global.DB.Where("session_id = ?", sessionID).Order("created_at asc").Find(&history).Error; err != nil {
		c.JSON(http.StatusInternalServerError, gin.H{"error": "获取历史记录失败: " + err.Error()})
		return
	}

	c.JSON(http.StatusOK, gin.H{
		"message": "获取成功",
		"data":    history,
	})
}
```

### 3. `chatMessageHandler` (核心：发送消息与 Agent 思考入口)

这个接口最关键。它的逻辑是：接收用户的提问 -> 判断是否是新会话 -> 记录用户消息入库 -> **触发大模型的 Agent 思考循环** -> 返回大模型的最终答案并入库。

Go

```
package handlers

import (
	"fmt"
	"io"
	"net/http"
	"time"

	"github.com/gin-gonic/gin"
	"github.com/google/uuid"
	
	// "your_project/services" // 引入我们刚刚写的 service
	// "your_project/models"
	// "your_project/global"
)

// ChatRequest 定义前端传过来的 JSON 结构
type ChatRequest struct {
	SessionID string `json:"session_id"`
	Content   string `json:"content" binding:"required"`
}

func ChatMessageHandler(c *gin.Context) {
	var req ChatRequest
	if err := c.ShouldBindJSON(&req); err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"error": "参数错误: " + err.Error()})
		return
	}

	sessionID := req.SessionID

	// 1. 会话逻辑与用户消息入库（同上一次设计）
	if sessionID == "" {
		sessionID = uuid.New().String()
		/*
		newSession := models.ChatSession{
			ID:    sessionID,
			Title: req.Content,
		}
		global.DB.Create(&newSession)
		*/
	} else {
		/*
		global.DB.Model(&models.ChatSession{}).Where("id = ?", sessionID).Update("updated_at", time.Now())
		*/
	}

	/*
	userMsg := models.ChatHistory{
		SessionID: sessionID,
		Role:      "user",
		Content:   req.Content,
	}
	global.DB.Create(&userMsg)
	*/

	// 2. 准备开启 Server-Sent Events (SSE) 流式传输
	// 设置必要的响应头，告诉前端这是一个持久化的流，不要缓存
	c.Writer.Header().Set("Content-Type", "text/event-stream; charset=utf-8")
	c.Writer.Header().Set("Cache-Control", "no-cache")
	c.Writer.Header().Set("Connection", "keep-alive")
	// 如果前端存在跨域，务必确保允许相关 Header

	// 先把 session_id 作为元数据推给前端，方便新建会话时前端能够立刻更新路由
	c.SSEvent("meta", gin.H{"session_id": sessionID})
	c.Writer.Flush()

	// 3. 创建一个消息通道，用于连接 Service 层和 Handler
	streamChan := make(chan services.StreamMessage)

	// 4. 开启独立的 Goroutine 触发 Agentic RAG 工作流
	go services.RunAgentLoop(sessionID, req.Content, streamChan)

	// 5. 监听通道，持续不断地将数据推送给客户端 (阻塞并监听)
	c.Stream(func(w io.Writer) bool {
		// 从通道阻塞读取数据
		if msg, ok := <-streamChan; ok {
			// 将接收到的数据转化为 SSE 格式推出
			// 最终的网络报文格式类似于:
			// event: status
			// data: 正在分析您的意图...
			c.SSEvent(msg.Event, msg.Data)
			return true // 返回 true 表示继续保持流打开
		}
		return false // 通道关闭，返回 false，Gin 会自动断开响应结束流
	})
}
```
